CONTROL: lblHorasLabMant   (NUEVO - Etiqueta: resultado del calculo)
-------------------------------------------------------------------------------
Esta es la formula de HORAS LABORALES. Respeta el horario:
  Lunes a Jueves .... 07:00 - 17:00  (10 h)
  Viernes ........... 07:00 - 16:00  ( 9 h)
  Sabado / Domingo .. solo cuenta si el trabajo INICIA o TERMINA ese dia;
                      como dia intermedio cuenta 0 h.

Text:
=With(
    {
        d1: dpIniRealMant.SelectedDate;
        d2: dpRealMant.SelectedDate;
        h1: varHIniN;
        h2: varHFinN
    };
    With(
        {
            c1: Switch(Weekday(d1; StartOfWeek.Monday); 5; 16; 6; 16; 7; 16; 17);
            c2: Switch(Weekday(d2; StartOfWeek.Monday); 5; 16; 6; 16; 7; 16; 17)
        };
        If(
            IsBlank(d1) || IsBlank(d2) || d2 < d1;
            "Horas laborales: 0,00 h  (revisa las fechas)";

            "Horas laborales: " &
            Text(
                If(
                    d1 = d2;
                    Max(0; Round(Min(c2; h2) - Max(7; h1); 2));
                    Round(
                        Max(0; c1 - Max(7; h1))
                        + Max(0; Min(c2; h2) - 7)
                        + Coalesce(
                            Sum(
                                ForAll(
                                    Sequence(Max(0; DateDiff(d1; d2; TimeUnit.Days) - 1); 1) As S;
                                    {
                                        H: Switch(
                                            Weekday(DateAdd(d1; S.Value; TimeUnit.Days); StartOfWeek.Monday);
                                            5; 9;
                                            6; 0;
                                            7; 0;
                                            10
                                        )
                                    }
                                );
                                H
                            );
                            0
                        );
                        2
                    )
                );
                "[$-es-GT]#,##0.00"
            ) & " h   (" &
            Text(d1; "dd/mm") & " " & Text(Int(h1); "[$-en-US]00") & ":" & Text(Round(Mod(h1; 1) * 60; 0); "[$-en-US]00") &
            "  a  " &
            Text(d2; "dd/mm") & " " & Text(Int(h2); "[$-en-US]00") & ":" & Text(Round(Mod(h2; 1) * 60; 0); "[$-en-US]00") & ")"
        )
    )
)

    /* ============ 2. ASIGNACION = FECHA + HORA (aqui arranca el reloj) ====== */
    Set(
        varFechaAsig;
        dpAsigMant.SelectedDate +
        If(
            !IsBlank(varRegMant.ASIGNACION)
              && DateValue(varRegMant.ASIGNACION) = dpAsigMant.SelectedDate;
            Time(Hour(varRegMant.ASIGNACION); Minute(varRegMant.ASIGNACION); 0);
            Time(Hour(varAhora); Minute(varAhora); 0)
        )
    );;

    /* ============ 3. ENTREGA ESTIMADA SEGUN PRIORIDAD ============ */
    Set(
        varFechaComp;
        DateAdd(
            varFechaAsig;
            Switch(
                cmbPrioridadMant.Selected.Value;
                "Urgente"; 1;
                "Alta 24 hrs"; 1;
                "Media 72 hrs"; 3;
                "Baja 1 semana"; 7;
                "Pendiente"; 30;
                30
            );
            TimeUnit.Days
        )
    );;

    /* ============ 4. INICIO / FIN REAL CON HORA MANUAL ============ */
    Set(
        varIniReal;
        If(
            varEstadoNuevo = "FINALIZADO";
            dpIniRealMant.SelectedDate
              + Time(Int(varHIniN); Round(Mod(varHIniN; 1) * 60; 0); 0);
            varRegMant.'INICIO REAL'
        )
    );;
    Set(
        varFinReal;
        If(
            varEstadoNuevo = "FINALIZADO";
            dpRealMant.SelectedDate
              + Time(Int(varHFinN); Round(Mod(varHFinN; 1) * 60; 0); 0);
            varRegMant.'FIN REAL'
        )
    );;

    /* ============ 5. HORAS LABORALES (L-J 7-17, V 7-16, S/D no cuentan) ==== */
    Set(
        varHorasTot;
        If(
            varEstadoNuevo <> "FINALIZADO";
            Coalesce(varRegMant.HORAS_TOTALES; 0);
            With(
                {
                    d1: dpIniRealMant.SelectedDate;
                    d2: dpRealMant.SelectedDate;
                    h1: varHIniN;
                    h2: varHFinN
                };
                With(
                    {
                        c1: Switch(Weekday(d1; StartOfWeek.Monday); 5; 16; 6; 16; 7; 16; 17);
                        c2: Switch(Weekday(d2; StartOfWeek.Monday); 5; 16; 6; 16; 7; 16; 17)
                    };
                    If(
                        d1 = d2;
                        Max(0; Round(Min(c2; h2) - Max(7; h1); 2));
                        Round(
                            Max(0; c1 - Max(7; h1))
                            + Max(0; Min(c2; h2) - 7)
                            + Coalesce(
                                Sum(
                                    ForAll(
                                        Sequence(Max(0; DateDiff(d1; d2; TimeUnit.Days) - 1); 1) As S;
                                        {
                                            H: Switch(
                                                Weekday(DateAdd(d1; S.Value; TimeUnit.Days); StartOfWeek.Monday);
                                                5; 9;
                                                6; 0;
                                                7; 0;
                                                10
                                            )
                                        }
                                    );
                                    H
                                );
                                0
                            );
                            2
                        )
                    )
                )
            )
        )
    );;

    /* ============ 6. HISTORIAL ============ */
    Set(
        varHist;
        Coalesce(varRegMant.HISTORIAL; "") &
        Text(varAhora; "dd/mm/yyyy hh:mm") & " - " & varEstadoNuevo & " - " & varUsuarioActual &
        If(varEstadoNuevo = "FINALIZADO"; " - " & Text(varHorasTot; "[$-es-GT]#,##0.00") & " h"; "") &
        If(IsBlank(txtComentMant.Text); ""; " - " & txtComentMant.Text) &
        Char(10)
    );;

    /* ============ 7. GUARDAR ORDEN ============ */
    Set(
        varIDOrden;
        Switch(
            varFuentePanel;

            "ACTIVO";
            Patch(
                DBASIG;
                Defaults(DBASIG);
                {
                    AREA: varRegMant.AREA;
                    CATEGORIA: varRegMant.CATEGORIA;
                    DESCRIPCION: varRegMant.DESCRIPCION;
                    ID_ACTIVO: varIDActivo;
                    SOLICITANTE: User().FullName;
                    CORREO: varUsuarioActual;
                    CREADO: varAhora;
                    RESPONSABLE: varRespTxt;
                    ASIGNACION: varFechaAsig;
                    COMPROMISO: varFechaComp;
                    'INICIO REAL': varIniReal;
                    'FIN REAL': varFinReal;
                    HORAS_TOTALES: varHorasTot;
                    PRIORIDADES: {Value: cmbPrioridadMant.Selected.Value};
                    ESTADO: {Value: varEstadoNuevo};
                    TIPOS: {Value: "INVERNADERO"};
                    TIPO: {Value: "PREVENTIVO"};
                    COMENTARIO: txtComentMant.Text;
                    HISTORIAL: varHist
                }
            ).ID;

            "REVMANT";
            Patch(
                REVISIONMANT;
                LookUp(REVISIONMANT; ID = varRegMant.ID);
                {
                    RESPONSABLE: varRespTxt;
                    ESTADO: {Value: varEstadoNuevo};
                    PRIORIDAD: {Value: cmbPrioridadMant.Selected.Value};
                    ASIGNACION: varFechaAsig;
                    COMPROMISO: varFechaComp;
                    INICIOREAL: varIniReal;
                    FINREAL: varFinReal;
                    HORAS_TOTALES: varHorasTot;
                    COMENTARIO: txtComentMant.Text;
                    ASIGNO: If(IsBlank(varRegMant.CORREO); User().FullName; varRegMant.SOLICITANTE);
                    CORREOASIGNO: Coalesce(varRegMant.CORREO; varUsuarioActual)
                }
            ).ID;

            Patch(
                DBASIG;
                LookUp(DBASIG; ID = varRegMant.ID);
                {
                    ESTADO: {Value: varEstadoNuevo};
                    PRIORIDADES: {Value: cmbPrioridadMant.Selected.Value};
                    RESPONSABLE: varRespTxt;
                    ASIGNACION: varFechaAsig;
                    COMPROMISO: varFechaComp;
                    'INICIO REAL': varIniReal;
                    'FIN REAL': varFinReal;
                    HORAS_TOTALES: varHorasTot;
                    COMENTARIO: txtComentMant.Text;
                    HISTORIAL: varHist;
                    SOLICITANTE: Coalesce(varRegMant.SOLICITANTE; User().FullName);
                    CORREO: Coalesce(varRegMant.CORREO; varUsuarioActual)
                }
            ).ID
        )
    );;

    /* ============ 8. PREVENTIVO FINALIZADO: REINICIA EL ACTIVO ============ */
    If(
        varTipoPanel = "PREVENTIVO" && varEstadoNuevo = "FINALIZADO" && varIDActivo > 0;
        Patch(
            DB;
            LookUp(DB; ID = varIDActivo);
            {
                INSTALACION: dpRealMant.SelectedDate;
                CAMBIO: DateAdd(
                    dpRealMant.SelectedDate;
                    Value(Coalesce(LookUp(DB; ID = varIDActivo).'VIDA UTIL'; 0));
                    TimeUnit.Days
                );
                HORAS_ACUM: 0;
                'DIAS RESTANTES': Value(Coalesce(LookUp(DB; ID = varIDActivo).'VIDA UTIL'; 0));
                ESTADOS: {Value: "VIGENTE"};
                ULTIMO_CALCULO: varAhora
            }
        )
    );;

    /* ============ 9. CORREO ============ */
    If(
        (varTipoPanel = "CORRECTIVO" && varEstadoNuevo in ["ASIGNADO"; "FINALIZADO"])
          || (varTipoPanel in ["PREVENTIVO"; "REVISION"] && varEstadoNuevo = "FINALIZADO");
        With(
            {
                dest: Coalesce(varRegMant.CORREO; varUsuarioActual);
                tipoTxt: Switch(

                dest;

                Coalesce(varRegMant.AREA; "") & "</b> cambio a estado <b style='color:" &
                Switch(varEstadoNuevo; "ASIGNADO"; "#e8a020"; "FINALIZADO"; "#16a34a"; "#0078d4") &
                "'>" & varEstadoNuevo & "</b>.</p>" &
                "<table style='border-collapse:collapse;font-size:13px;min-width:420px'>" &
                Concat(
                    Table(
                        {k: "Tipo"; v: tipoTxt};
                        {k: "Area"; v: Coalesce(varRegMant.AREA; "—")};
                        {k: "Descripcion"; v: Coalesce(varRegMant.DESCRIPCION; "—")};
                        {k: "Responsable"; v: varRespTxt};
                        {k: "Prioridad"; v: cmbPrioridadMant.Selected.Value};
                        {k: "Asignacion"; v: Text(varFechaAsig; "dd/mm/yyyy hh:mm")};

                                "—"

                ) &
                "</table>" &
                "<p style='font-size:12px;color:#93a4b4;margin-top:14px'>Mensaje automatico &middot; Equipo de Mantenimiento</p></div>";
                {IsHtml: true}

    If(varFuentePanel = "REVMANT"; Refresh(REVISIONMANT));;
    If(varTipoPanel = "PREVENTIVO"; Refresh(DB));;

        Switch(

                          If(varTipoPanel = "PREVENTIVO"; ". El activo reinicio su fecha de cambio."; "");
            "Actualizado correctamente."
        );
        NotificationType.Success
    )
)

function main(
  workbook: ExcelScript.Workbook,
  hoja: string,
  datosJson: string
) {
  const datos = JSON.parse(datosJson) as Record<string, string | number | boolean>[];

  let sheet = workbook.getWorksheet(hoja);
  if (!sheet) {
    sheet = workbook.addWorksheet(hoja);
  }

  // limpiar tablas y contenido previo
  for (const t of sheet.getTables()) {
    t.delete();
  }
  const usado = sheet.getUsedRange();
  if (usado) {
    usado.clear(ExcelScript.ClearApplyTo.all);
  }

  if (datos.length === 0) {
    sheet.getRange("A1").setValue("Sin registros en el rango seleccionado.");
    return;
  }

  const encabezados: string[] = Object.keys(datos[0]);
  const matriz: (string | number | boolean)[][] = [];
  matriz.push(encabezados);

  for (const fila of datos) {
    const r: (string | number | boolean)[] = [];
    for (const h of encabezados) {
      const v = fila[h];
      r.push(v === null || v === undefined ? "" : v);
    }
    matriz.push(r);
  }

  const rango = sheet.getRangeByIndexes(0, 0, matriz.length, encabezados.length);
  rango.setValues(matriz);

  const cab = sheet.getRangeByIndexes(0, 0, 1, encabezados.length);
  cab.getFormat().getFill().setColor("#092162");
  cab.getFormat().getFont().setColor("#FFFFFF");
  cab.getFormat().getFont().setBold(true);

  const tabla = workbook.addTable(rango, true);
  tabla.setName("T_" + hoja);

  rango.getFormat().autofitColumns();
  sheet.getFreezePanes().freezeRows(1);
}

PRUEBA RAPIDA DEL CALCULO DE HORAS
-------------------------------------------------------------------------------
Caso A: inicio Lunes 14:00, fin Lunes 16:30        -> 2,50 h
Caso B: inicio Lunes 14:00, fin Martes 10:00       -> 3,00 h  (3 + 0 + 0... ver)
        Lunes 14:00 a 17:00 = 3 h ; Martes 07:00 a 10:00 = 3 h  -> 6,00 h
Caso C: inicio Viernes 14:00, fin Lunes 10:00      -> 2 h (V) + 0 (S) + 0 (D)
                                                      + 3 h (L) = 5,00 h
Caso D: inicio Lunes 07:00, fin Jueves 17:00       -> 10 + 10 + 10 + 10 = 40,00 h
Caso E: inicio Sabado 08:00, fin Sabado 12:00      -> 4,00 h  (si se trabaja
                                                      ese sabado, cuenta)

Si tu horario real cambia, los unicos numeros que hay que tocar son los del
Switch de Weekday, y aparecen en DOS lugares: lblHorasLabMant.Text y
btnGuardar.OnSelect (paso 5). Significado:
   Switch(Weekday(d; StartOfWeek.Monday); 5; 16; 6; 16; 7; 16; 17)
     -> hora de CIERRE: viernes 16, sabado 16, domingo 16, resto 17
   Switch(Weekday(...); 5; 9; 6; 0; 7; 0; 10)
     -> jornada COMPLETA de un dia intermedio: viernes 9 h, sabado 0,
        domingo 0, resto 10 h
La hora de APERTURA (07:00) esta escrita como el numero 7 en los Max(7; h1)
y Min(c2; h2) - 7.
